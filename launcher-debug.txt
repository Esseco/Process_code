[CmdletBinding()]
param([int]$Port = 8765)

$ErrorActionPreference = 'Stop'
$root = Split-Path -Parent $MyInvocation.MyCommand.Path
$server = Join-Path $root 'search_tool_server.ps1'
$url = "http://127.0.0.1:$Port"

function Test-Service {
    try {
        $response = Invoke-WebRequest -UseBasicParsing -Uri "$url/api/health" -TimeoutSec 1
        return $response.StatusCode -eq 200
    } catch { return $false }
}

if (-not (Test-Service)) {
    Start-Process -FilePath 'powershell.exe' -ArgumentList @('-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', $server, '-Port', $Port) -WorkingDirectory $root -WindowStyle Minimized | Out-Null
    for ($i = 0; $i -lt 20 -and -not (Test-Service); $i++) { Start-Sleep -Milliseconds 250 }
}

if (-not (Test-Service)) {
    Write-Host "搜索服务启动失败，请直接运行 search_tool_server.ps1 查看错误。"
    exit 1
}

Start-Process "$url/"
